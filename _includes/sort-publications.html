{% comment %} Sorts publications by `date` descending. Same-date ties are ordered by the time in `date` (e.g. 2026-09-25 12:00:00), then by filename — the same key publication-card.html uses for numbering. Usage: {% include sort-publications.html pubs=site.publications %} Result is assigned to `sorted_pubs`. {% endcomment %} {% assign sort_keys = "" | split: "" %} {% for p in include.pubs %} {% capture k %}{{ p.date | date: "%Y-%m-%d %H:%M:%S" }}|{{ p.id }}{% endcapture %} {% assign sort_keys = sort_keys | push: k %} {% endfor %} {% assign sort_keys = sort_keys | sort | reverse %} {% assign sorted_pubs = "" | split: "" %} {% for k in sort_keys %} {% assign kid = k | split: "|" | last %} {% for p in include.pubs %} {% if p.id == kid %}{% assign sorted_pubs = sorted_pubs | push: p %}{% break %}{% endif %} {% endfor %} {% endfor %}
